cmake_minimum_required(VERSION 3.22)
project(ComposerContracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Network Widgets)
add_executable(composer_contracts contracts.cpp ../internal/composerpackage.cpp ../internal/composersession.cpp ../internal/composersession.h)
target_compile_definitions(composer_contracts PRIVATE PIANOMANIA_COMPOSER_PROFILE="development" PIANOMANIA_COMPOSER_SERVICE_ENVIRONMENT="development" PIANOMANIA_COMPOSER_SERVICE_URL="http://127.0.0.1:5038")
target_link_libraries(composer_contracts PRIVATE Qt6::Core Qt6::Network Qt6::Widgets)
if(WIN32)
    target_link_libraries(composer_contracts PRIVATE Advapi32)
elseif(APPLE)
    target_link_libraries(composer_contracts PRIVATE "-framework Security")
endif()
enable_testing()
add_test(NAME composer_contracts COMMAND composer_contracts ${CMAKE_CURRENT_BINARY_DIR}/contract.pm)
add_subdirectory(../tests/composeraccount account)
add_subdirectory(../tests/composerbackground background)
